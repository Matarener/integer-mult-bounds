def compile_source(data, filename):
    return compile(data, str(filename), "exec", dont_inherit=True, optimize=0)


def source_module(name, path):
    """Execute precisely these source bytes; never import local bytecode caches."""
    module = types.ModuleType(name)
    module.__file__ = str(path)
    sys.modules[name] = module
    exec(compile_source(path.read_bytes(), path), module.__dict__)
    return module

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def combine(*terms):
    out = Counter()
    for c, coefficients in terms:
        for r, n in coefficients.items():
            out[r] += c * n
    return {r: Q(n) for r, n in sorted(out.items()) if n}


def local_histogram(k):
    need(0 <= k <= N4, "rank-four donor inventory")
    return combine((CALLS, H), (k, {3: 1, 7: -1}))


def relaxed_idle(k):
    return {2: Q(67872, 5), 4: Q(12768 + k), 6: Q(14208),
            7: Q(21312), 42: Q(3168, 5)}


def relaxed_histogram(k):
    return combine((1, local_histogram(k)), (1, relaxed_idle(k)))


def raw_excess(histogram, b):
    return sub(linear_phi(histogram, b), exact(M * W))


def discounted_excess(histogram, b):
    """Grant every bad packet free and retain only the uniform good fraction."""
    return sub(mul(linear_phi(histogram, b), 1-EPSILON), exact(M*W))


def validate_primal(p, k):
    need(all(n >= 0 for n in p.values()), "nonnegative primal counts")
    for R, count in BIN_COUNTS.items():
        need(sum(n for (r, _, _), n in p.items() if r == R) == count,
             "all bins accounted")
    need(sum(a * n for (_, a, _), n in p.items()) == N8, "all rank-eight helpers")
    need(sum(b * n for (_, _, b), n in p.items()) == N4-k, "all retained rank-four helpers")
    idle = Counter()
    for (R, a, b), n in p.items():
        q = R - 8 * a - 4 * b
        need(q >= 0, "pattern capacity")
        if q:
            idle[q] += n
    need(dict(idle) == relaxed_idle(k), "primal idle histogram")


def primal(k):
    """Fractional bin pattern counts (R, full-eight helpers, partial-fours)."""
    n4 = N4 - k
    p = {
        (7, 0, 0): Q(21312),
        (14, 1, 0): Q(14208),
        (28, 3, 0): Q(14208 - n4),
        (28, 3, 1): Q(n4),
        (42, 5, 0): Q(67872, 5),
        (42, 0, 0): Q(3168, 5),
    }
    validate_primal(p, k)
    return [{"R": R, "full8": a, "partial4": b, "count": str(n)}
            for (R, a, b), n in sorted(p.items())]


def dual_coefficients(R, a, b):
    """Coefficients of Phi(q)-lambda8*a-lambda4*b-mu_R."""
    q = R - 8 * a - 4 * b
    l8 = {2: Q(1, 5), 42: -Q(1, 5)}
    l4 = {4: Q(-1)}
    mus = {7: {7: Q(1)},
           14: combine((1, {6: 1}), (-1, l8)),
           28: combine((1, {4: 1}), (-3, l8)),
           42: {42: Q(1)}}
    return combine((1, {q: 1} if q else {}), (-a, l8), (-b, l4), (-1, mus[R]))


def certify_dual(b):
    """Every feasible integer occupancy pattern is covered, including rank zero."""
    result, equality = [], []
    for R in BIN_COUNTS:
        for a in range(R // 8 + 1):
            for p in range((R - 8 * a) // 4 + 1):
                coefficients = dual_coefficients(R, a, p)
                if not coefficients:
                    equality.append([R, a, p])
                    continue
                interval = linear_phi(coefficients, b)
                need(interval[0] > 0, f"strict dual feasibility R={R},a={a},p={p}")
                result.append({"pattern": [R, a, p], "slack": report(interval)})
    # Every positive primal column is an exact equality column, for all k.
    for k in (0, N4):
        for row in primal(k):
            if Q(row["count"]) > 0:
                need(not dual_coefficients(row["R"], row["full8"], row["partial4"]),
                     "complementary slackness")
    l8 = linear_phi({2: Q(1, 5), 42: -Q(1, 5)}, b)
    l4 = mul(phi(4, b), -1)
    bank8, bank4 = sub(mul(l8, -1), exact(8)), sub(mul(l4, -1), exact(4))
    need(bank8[0] > 0 and bank4[0] > 0, "full-progress banks cannot improve the lower bound")
    discounted_bank8 = sub(mul(l8, -(1-EPSILON)), exact(8))
    discounted_bank4 = sub(mul(l4, -(1-EPSILON)), exact(4))
    need(discounted_bank8[0] > 0 and discounted_bank4[0] > 0,
         "discounted public dual still pays for unit-priced bank completions")
    return {"b": str(b), "patterns_checked": len(result) + len(equality),
            "exact_equality_patterns": equality, "strict_slacks": result,
            "extra_full_progress_bank_penalties":
                {"per_full8": report(bank8), "per_partial4": report(bank4),
                 "discounted_public_per_full8": report(discounted_bank8),
                 "discounted_public_per_partial4": report(discounted_bank4),
                 "premise": "extra bank completions retain undiscounted unit-mass cost"}}


def all_parameter_dual_controls():
    """Integer-product endpoints for the two four-power inequalities in PROOF.md."""
    rows = {
        "rank14_chord": [(2, 1), (6, -5), (14, 5), (42, -1)],
        "rank28_chord": [(2, 3), (4, -5), (28, 5), (42, -3)],
    }
    result = {}
    for name, row in rows.items():
        need(sum(c for _, c in row) == 0, "power polynomial vanishes at zero")
        need(sum(r*c for r,c in row) == 0, "power polynomial vanishes at one")
        def ratio(weights):
            return Q(prod(r**c for r,c in weights if c > 0),
                     prod(r**(-c) for r,c in weights if c < 0))
        d0, d1 = ratio(row), ratio([(r, r*c) for r,c in row])
        need(d0 > 1 and d1 < 1, "strict derivative endpoint signs")
        result[name] = {"coefficients": row, "exp_derivative_at_zero": str(d0),
                        "exp_derivative_at_one": str(d1),
                        "zero_count_bound": 3,
                        "written_consequence": "strictly positive for every 0<p<1"}
    return result


def parametric_resource_controls():
    """Coefficient-wise resource identities, not a numerical parameter search."""
    unit = lambda name: {name: Q(1)}
    c7, c14, c28, c42, full, partial, donated = map(
        unit, ("C7", "C14", "C28", "C42", "F", "D", "k"))
    t = combine((Q(1, 5), full), (-Q(1, 5), c14), (-Q(3, 5), c28))
    rows = {
        (7, 0, 0): c7,
        (14, 1, 0): c14,
        (28, 3, 0): combine((1, c28), (-1, partial)),
        (28, 3, 1): partial,
        (42, 5, 0): t,
        (42, 0, 0): combine((1, c42), (-1, t)),
    }
    for R in (7, 14, 28, 42):
        got = combine(*[(1, count) for (r, _, _), count in rows.items() if r == R])
        need(got == unit("C"+str(R)), "parametric bin identity")
    need(combine(*[(a, count) for (_, a, _), count in rows.items()]) == full,
         "parametric full-helper identity")
    need(combine(*[(d, count) for (_, _, d), count in rows.items()]) == partial,
         "parametric partial-helper identity")
    need(all(not dual_coefficients(R, a, d) for R,a,d in rows),
         "all parametric primal patterns are exact dual contacts")
    idle = {}
    for (R, a, d), count in rows.items():
        remainder = R-8*a-4*d
        if remainder:
            idle[remainder] = combine((1, idle.get(remainder, {})), (1, count))
    expected = {7: c7, 6: c14, 4: combine((1, c28), (-1, partial)),
                2: t, 42: combine((1, c42), (-1, t))}
    need(idle == expected, "parametric remainder histogram")
    need(combine(*[(r, count) for r,count in idle.items()]) ==
         combine((7, c7), (14, c14), (28, c28), (42, c42),
                 (-8, full), (-4, partial)), "parametric idle mass identity")
    # Substitute D -> D-k and append the genuine COPY change k(e3-e7).
    delta = {}
    for r, count in idle.items():
        changed = combine((1, count), (-count.get("D", 0), donated))
        difference = combine((1, changed), (-1, count))
        if difference:
            delta[r] = difference
    delta[3] = combine((1, delta.get(3, {})), (1, donated))
    delta[7] = combine((1, delta.get(7, {})), (-1, donated))
    need(delta == {3: donated, 4: donated, 7: {"k": Q(-1)}},
         "parametric complete donation delta")
    actual = {"C7": Q(21312), "C14": Q(14208), "C28": Q(14208),
              "C42": Q(14208), "F": Q(N8), "D": Q(N4)}
    need({r: sum(c*actual[v] for v,c in count.items()) for r,count in idle.items()} ==
         relaxed_idle(0), "actual inventory is a specialization")
    encode = lambda form: {v: str(c) for v,c in sorted(form.items())}
    return {
        "domain": ["C7,C14,C28,C42>=0", "0<=D<=C28",
                   "C14+3*C28<=F<=C14+3*C28+5*C42",
                   "0<=k<=D; at least k genuine rank-seven COPY jobs"],
        "t": encode(t),
        "primal": [{"R": R, "full8": a, "partial4": d, "count": encode(count)}
                   for (R,a,d), count in sorted(rows.items())],
        "idle_histogram": {str(r): encode(count) for r,count in sorted(idle.items())},
        "complete_donation_delta": {str(r): encode(count) for r,count in sorted(delta.items())},
        "identities": ["all bins", "F full helpers", "D partial helpers", "idle mass",
                       "all primal patterns dual-tight", "D-to-D-k substitution",
                       "actual nine-stage inventory specialization"],
        "scope": "fractional counts of whole legal bin patterns; all 0<b<1; no new caller",
    }


def actual_public_bins():
    """Extract the pinned recipe and idle insertion; preserve every source guard."""
    path = CONSUMER / "inputs/retained/source-context/global_k.py"
    need(digest(path) == "6368196a393ac25bb2918880e4c4bd31ed733b762c27b6873ab391cbc0c5c3c8",
         "public controller source")
    tree = ast.parse(path.read_bytes())
    recipe = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "recipe")
    blocks = [n for n in ast.walk(tree) if isinstance(n, ast.If) and
              ast.unparse(n.test) == "s % 2 == 1 and s <= 2 * K - 3"]
    need(len(blocks) == 1, "unique source idle insertion")
    class KeepChecks(ast.NodeTransformer):
        def visit_Assert(self, node):
            return ast.copy_location(ast.Expr(value=ast.Call(
                func=ast.Name(id="need", ctx=ast.Load()),
                args=[node.test, ast.Constant(value="source controller guard")],
                keywords=[])), node)
    module = ast.fix_missing_locations(ast.Module(body=[recipe], type_ignores=[]))
    env = {"__builtins__": {"range": range, "reversed": reversed, "dict": dict}}
    exec(compile(module, "pinned-recipe", "exec"), env)
    block = ast.fix_missing_locations(KeepChecks().visit(
        ast.Module(body=[blocks[0]], type_ignores=[])))
    code = compile(block, "pinned-idle-insertion", "exec")
    levels, schedule = [-1]*4, []
    for event in env["recipe"](4):
        if event["kind"] == "local":
            s, p = event["stage"], event["pair"]
            need(levels[p] == s-1, "source local entrance")
            levels[p] = s
            exec(code, {"__builtins__": {"range": range, "dict": dict}, "need": need,
                        "K": 4, "s": s, "levels": levels, "schedule": schedule})
        elif event["kind"] == "bridge":
            need(levels[event["dst"]] == levels[event["src"]], "source bridge entrance")
        else:
            need(event["kind"] == "exchange" and levels == [8]*4, "source exchange entrance")
    ranks = [e["rank"] for e in schedule]
    need(ranks == [14, 7, 28, 7, 42, 7, 42, 28, 14, 0], "all public idle events")
    bins = Counter()
    for rank in ranks:
        if rank:
            bins[rank] += 2 * 111 * 32
    need(dict(bins) == BIN_COUNTS, "public bin multiplicities from the source recipe")
    return {"source_sha256": digest(path), "idle_sequence": ranks,
            "bins": dict(sorted(bins.items())), "source_extraction_only": True}


def source_controls():
    for name, pin in PINS.items():
        need(digest(HERE / name) == pin, f"source hash {name}")
    wp = CONSUMER / "inputs/retained/boundary/inputs/WORD.json.gz"
    need(digest(wp) == "c46ccd7d3eedabc6a7efbb1a91ec13f2928bad8ee0bac57c6e593a092d13d96c",
         "frozen actual local word")
    word = json.loads(gzip.decompress(wp.read_bytes()))
    gp = wp.with_name("GEOMETRY.json.gz")
    need(digest(gp) == "a353999bceac4ac957f540adfd3e5f8768297138b27132eeb83332e2ea4b1022",
         "actual frame geometry")
    geometry = json.loads(gzip.decompress(gp.read_bytes()))
    need(geometry["v"] == 111 and geometry["h"] == 8, "source public ports and local width")
    def rank_of_frame(frame):
        pivots = {}
        for label in geometry["frames"][frame]:
            row = list(map(Q, geometry["lines"][label]))
            for pivot, basis in sorted(pivots.items()):
                coefficient = row[pivot]
                row = [x-coefficient*y for x,y in zip(row, basis)]
            pivot = next((i for i,x in enumerate(row) if x), None)
            if pivot is not None:
                scale = row[pivot]
                pivots[pivot] = [x/scale for x in row]
            if len(pivots) == 8:
                break
        return len(pivots)
    dims = {f: rank_of_frame(f) for f in set(word["final"][222:660]) | set(range(510,519)) | {0}}
    need(all(word["start"][h] == 0 for h in range(222,660)) and dims[0] == 0,
         "all helpers enter at zero progress")
    widths = Counter(dims[f] for f in word["final"][222:660])
    need(widths == {4: 5, 8: 433}, "actual helper width inventory from frame geometry")
    paid = Counter()
    copies = 0
    for op, d, s, c, f, phase in word["records"]:
        if op == 0 and f:
            paid[f] += 1
        elif op == 2:
            need(c in range(510, 519) and f == 0 and s == 660,
                 "authenticated rank-seven COPY endpoint")
            need(dims[c] - dims[f] == 7, "COPY rank from actual geometry")
            paid[7] += 1
            copies += 1
    need(dict(paid) == H and copies == 9, "entire actual local query histogram")
    need(sum(H.values()) == 2913 and sum(r*n for r,n in H.items()) == 5101,
         "local count and rank mass")
    need(sum(R * n for R, n in BIN_COUNTS.items()) == 1342656, "public idle stock")
    need(N8 == 124704 and N4 == 1440, "whole helper inventories")
    need(sum(r*n for r,n in relaxed_histogram(0).items()) == 1808352, "full mass")
    need(M*W - 1808352 == 10272, "complete deficit")
    return {"word_sha256": digest(wp), "geometry_sha256": digest(gp),
            "helper_widths": dict(widths), "paid_local_histogram": H,
            "public_controller": actual_public_bins(),
            "copies": copies, "local_calls": CALLS, "public_families": W,
            "private_reserve_families": 9}

        args.output.parent.mkdir(parents=True, exist_ok=True)
        args.output.write_text(data)
    else:
        print(data, end="")


if __name__ == "__main__":
    main()
