"""One unprofiled PR5818/candidate pair after the small feasibility gate."""
from pathlib import Path
import subprocess
import sys
import json

HERE = Path(__file__).resolve().parent
BASE = HERE.parent/'single-pass-qualification-20261027/repo/research/selected-single-pass-qualification'
for label, script in [('baseline', BASE/'joint.py'), ('candidate', HERE/'candidate-run.py')]:
    completed = subprocess.run([sys.executable,'-B',str(HERE/'guarded.py'),label,str(script)])
    if completed.returncode:
        raise SystemExit(completed.returncode)
old = json.loads((HERE/'runs/baseline/receipt.json').read_bytes())
new = json.loads((HERE/'runs/candidate/receipt.json').read_bytes())
cpu = lambda r: r['usage']['user_seconds'] + r['usage']['system_seconds']
result = {'baseline_wall_seconds':old['wall_seconds'],'candidate_wall_seconds':new['wall_seconds'],
          'baseline_cpu_seconds':cpu(old),'candidate_cpu_seconds':cpu(new),
          'wall_fraction_saved':1-new['wall_seconds']/old['wall_seconds'],
          'cpu_fraction_saved':1-cpu(new)/cpu(old),
          'complete_stdout_equal':old['stdout_sha256']==new['stdout_sha256'],
          'profile_included_in_comparison':False,'samples_per_arm':1}
(HERE/'COMPARISON.json').write_text(json.dumps(result,indent=2,sort_keys=True)+'\n')
if not result['complete_stdout_equal'] or min(result['wall_fraction_saved'],result['cpu_fraction_saved'])<.15:
    raise ValueError('The required full-coverage 15 percent wall and CPU gate failed')
print('PASS_LITERAL_SETUP_FIRST_PAIR',json.dumps(result,sort_keys=True))
